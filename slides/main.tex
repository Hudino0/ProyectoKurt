\section{Fundamentos de Machine Learning}
%============================================================

\begin{frame}{¿Qué es un problema de regresión supervisada?}
    \justifying
    Los tres métodos del artículo resuelven la misma tarea: una \textbf{regresión supervisada}.

    \begin{itemize}
        \item Los datos de entrenamiento tienen las entradas (tipo de motor, condiciones ambientales, etc.) y el valor real de la salida (\textit{fuel flow T/O}).
        \item El algoritmo \textbf{aprende} una función $f$. La predicción $\hat{y} = f(x_1, x_2, \dots, x_n)$ debe quedar cerca del valor real $y$.
        \item La salida es un \textbf{número continuo} (kg/s), no una categoría. Por eso el problema es de regresión y no de clasificación.
    \end{itemize}
    \bigskip

    \textbf{GPR}, \textbf{SVM} y \textbf{MLP} construyen la función $f$ de tres formas distintas.
\end{frame}

\subsection{Regresión de Procesos Gaussianos (GPR)}

\begin{frame}{GPR: la idea intuitiva}
    \justifying
    La \textbf{Regresión de Procesos Gaussianos} (\textit{Gaussian Process Regression}, GPR) no ajusta una sola curva. Trabaja con una \textbf{distribución de probabilidad sobre funciones}.

    \begin{itemize}
        \item Antes de ver los datos, el modelo acepta \textbf{muchas} funciones posibles.
        \item Cada dato observado \textbf{descarta} las funciones que no pasan cerca de él.
        \item El resultado es una función media (la predicción) y una \textbf{banda de incertidumbre}. La banda indica qué tan confiable es la predicción en cada zona.
    \end{itemize}
\end{frame}

\begin{frame}{GPR: función media, incertidumbre y kernel}
    \begin{columns}[c]
        \column{5.6cm}
        \small
        Definición (Ec.~3 del artículo):
        $$ f(x) \sim \text{GPR}\big(m(x), k(x,x')\big) $$
        \begin{itemize}
            \item $m(x)$: función \textbf{media}. Es la predicción puntual.
            \item $k(x,x')$: función de \textbf{covarianza} (\textit{kernel}). Mide qué tan parecidos son dos puntos $x$ y $x'$. Define qué tan suave es la función.
        \end{itemize}
        \bigskip
        El artículo prueba 3 \textit{kernels}: \textbf{exponencial}, \textbf{exponencial cuadrado} y \textbf{racional cuadrático}.
        \column{5.6cm}
        \begin{tikzpicture}[scale=0.78]
        \begin{axis}[

            \addplot[suda, thick, domain=0:6, samples=60] {1.6+0.4*sin(deg(1.3*x))};

    \end{columns}
\end{frame}

\subsection{Máquinas de Vectores de Soporte (SVM)}

\begin{frame}{SVM: la idea intuitiva}
    \justifying\small
    Las \textbf{Máquinas de Vectores de Soporte} (\textit{Support Vector Machines}, SVM) nacieron para \textbf{clasificar}: separan dos grupos con la mejor frontera. También sirven para \textbf{regresión}: buscan la función que mejor sigue a los datos.
    \medskip

    \begin{itemize}
        \item El modelo define un \textbf{tubo} de ancho $\varepsilon$ alrededor de la función.
        \item Los puntos \textbf{dentro} del tubo no generan error. El modelo los trata como ruido normal.
        \item Los puntos \textbf{fuera} del tubo definen la forma de la función. Son los \textbf{vectores de soporte}.
    \end{itemize}
    \medskip
    \begin{alertblock}{Nota sobre el nombre}
        En regresión, el nombre usual es SVR (\textit{Support Vector Regression}). El artículo usa SVM, y esta presentación también.
    \end{alertblock}
\end{frame}

\begin{frame}{SVM: el tubo $\varepsilon$-insensible y los vectores de soporte}
    \begin{columns}[c]
        \column{5.6cm}
        \begin{tikzpicture}[scale=0.78]
        \begin{axis}[
            width=6.6cm, height=5.3cm,
            xlabel={$x$}, ylabel={$y$},
            xtick=\empty, ytick=\empty,
            axis lines=left,
            ]
            \addplot[name path=up, draw=none, domain=0:6] {1.0+0.28*x+0.45};
            \addplot[name path=lo, draw=none, domain=0:6] {1.0+0.28*x-0.45};
            \addplot[suda!18] fill between[of=up and lo];
            \addplot[suda, thick, domain=0:6] {1.0+0.28*x};
            \addplot[only marks, mark=*, black, mark size=1.6pt] coordinates {(0.5,1.3)(1.6,1.55)(2.5,1.75)(3.6,2.0)(4.7,2.15)};
            \addplot[only marks, mark=o, draw=suda, thick, mark size=2.4pt] coordinates {(1.0,2.35)(3.0,1.15)(5.2,2.9)};
        \end{axis}
        \end{tikzpicture}
        \column{5.6cm}
        \footnotesize
        \begin{itemize}
            \item \textbf{Puntos negros}: están dentro del tubo $\pm\varepsilon$. No generan error.
            \item \textbf{Círculos rojos}: están fuera del tubo. Son los \textbf{vectores de soporte}.
            \item Si la relación no es lineal, una \textbf{función kernel} lleva los datos a un espacio donde sí es lineal (\textit{kernel trick}).
        \end{itemize}
        \footnotesize\textit{Esquema ilustrativo.} El artículo prueba los \textit{kernels} \textbf{lineal}, \textbf{cuadrático} y \textbf{cúbico}.
    \end{columns}
\end{frame}

\subsection{Perceptrón Multicapa (MLP)}

\begin{frame}{MLP: la neurona artificial y la red}
    \justifying
    El \textbf{Perceptrón Multicapa} (\textit{Multilayer Perceptron}, MLP) es una \textbf{red neuronal artificial}. Su estructura imita, de forma simple, las conexiones entre neuronas biológicas.

    \begin{itemize}
        \item Cada \textbf{neurona} multiplica sus entradas por \textbf{pesos}, las suma y aplica una \textbf{función de activación} (p.~ej. \textit{tansig}, \textit{logsig}, \textit{purelin}).
        \item Las neuronas forman \textbf{capas}: una capa de entrada, una o más capas \textbf{ocultas} y una capa de salida.
        \item El modelo \textbf{aprende} con \textbf{retropropagación} (\textit{backpropagation}): compara la predicción con el valor real y corrige los pesos desde la salida hacia la entrada.
    \end{itemize}
\end{frame}

    \begin{center}

        \item Activaciones: \textit{tansig} -- \textit{logsig} -- \textit{tansig} -- \textit{purelin}.

\begin{frame}{¿Por qué conservar las 7 variables?}
    \justifying\small
    El artículo no justifica esta decisión. Hay tres razones técnicas que la apoyan:
    \begin{enumerate}
        \item La regresión lineal solo mide efectos \textbf{lineales}. \textit{Press ratio} no tiene efecto lineal propio, pero puede aportar información a un modelo no lineal (GPR, SVM, MLP).
        \item La prueba ANOVA evalúa el \textbf{conjunto} de variables, no cada variable sola. El conjunto es significativo (Sig. $< 0.001$).
        \item Las 7 variables son parámetros físicos del motor y del ensayo de certificación. Todas tienen sentido físico para el consumo de combustible.
    \end{enumerate}
    \begin{alertblock}{Límite de confiabilidad}
        El artículo no limpia los valores atípicos ni prueba los modelos sin \textit{press ratio}. En la réplica de este proyecto, con los datos corregidos del EEDB, la regresión da el mismo resultado: $R^2 = 0.980$ y $t \approx 0$ para \textit{press ratio}. Los valores atípicos no cambian la conclusión de la regresión.
    \end{alertblock}
\end{frame}

\begin{frame}{La ecuación de regresión múltiple resultante}
    \small
    $$ y = \beta_0 + \beta_1 p_1 + \beta_2 p_2 + \cdots + \beta_n p_n + \epsilon \quad \text{(Ec.~1)} $$
    Con los coeficientes de la Tabla 9, la ecuación queda (Ec.~2):
    \begin{align*}
        \text{Fuel Flow } T/O = \ & 1.302 - 0.49 \times \text{tipo de motor} - 0.1 \times \text{bypass ratio} \\
        & - 7.308\times10^{-5} \times \text{press ratio} + 0.009 \times \text{rated output} \\
        & + 2.242\times10^{-7} \times \text{ambient baro} - 0.002 \times \text{ambient temp} \\
        & - 4.612 \times \text{ambient humidity}
    \end{align*}
    \footnotesize\textbf{Lectura de los signos:}
    \begin{itemize}
        \item Más \textit{rated output} (motor más potente) $\Rightarrow$ más flujo de combustible ($\beta>0$).
        \item Más \textit{bypass ratio} (motor más eficiente) $\Rightarrow$ menos flujo de combustible ($\beta<0$).
    \end{itemize}
    Esta regresión \textbf{no es} uno de los tres modelos de ML. Solo sirve para elegir las entradas.
\end{frame}

\begin{frame}{Métricas para comparar los modelos}
    \justifying\small
    Los tres modelos se comparan con tres métricas de error. Un valor menor es mejor. $n$ es el número de observaciones, $FF_{T/O}$ el valor real y $ff_{T/O}$ el valor predicho.
    \begin{align*}
        \text{MSE} &= \frac{1}{n}\sum_{i=1}^n (FF_{T/O} - ff_{T/O})^2 & \text{(Ec.~5, error cuadrático medio)}\\
        \text{MAE} &= \frac{1}{n}\sum_{i=1}^n |FF_{T/O} - ff_{T/O}| & \text{(Ec.~6, error absoluto medio)}\\
        \text{MAPE} &= \frac{100}{n}\sum_{i=1}^n \left| \frac{FF_{T/O} - ff_{T/O}}{FF_{T/O}} \right| & \text{(Ec.~7, error porcentual absoluto medio)}
    \end{align*}
    \begin{alertblock}{Criterio de calidad (Lewis, 1982)}
        Un modelo con \textbf{MAPE $< 10\,\%$} es un modelo ``muy bueno'' (de alta precisión).
    \end{alertblock}
\end{frame}

\section{Conclusiones y cierre}
%============================================================

\begin{frame}{Hallazgos principales}
    \justifying
    \begin{itemize}
        \item El \textbf{MLP} (modelo 1, \textit{trainlm}) es el mejor modelo global. Tiene el menor error en validación y prueba, y un MAPE menor que 5\,\% en todos los conjuntos.
        \item El \textbf{GPR exponencial} tiene el mejor ajuste en entrenamiento, pero más error que el MLP en validación y prueba.
        \item El \textbf{SVM cuadrático} es el mejor de los tres SVM, pero tiene el mayor error de los tres modelos finales.
        \item El MLP \textbf{generaliza bien} a los datos de 2021: MAPE de 2.90\,\%.
        \item Los \textbf{intervalos de confianza del 99\,\%} señalan una posible degradación sin desmontar el motor.
    \end{itemize}
\end{frame}

\begin{frame}{Aporte práctico del estudio}
    \justifying
    \begin{block}{¿Qué permite este modelo?}
        Evaluar el desempeño del motor en el despegue \textbf{sin desmontarlo} y \textbf{sin un experto}. Solo usa datos públicos del EEDB.
    \end{block}
    \bigskip
    El modelo puede ayudar a reducir:
    \begin{itemize}
        \item El consumo de combustible,
        \item Los costos de mantenimiento, porque detecta la degradación a tiempo,
        \item El costo del vuelo,
        \item El riesgo de accidentes por fallas no detectadas,
        \item Las emisiones contaminantes.
    \end{itemize}
\end{frame}

\begin{frame}{Discusión crítica (reflexión propia)}
    \justifying\small
    \begin{alertblock}{Fortalezas}
        Compara 3 tipos de modelos con la misma partición. Prueba el mejor modelo con datos de 2021. Los intervalos de confianza son fáciles de interpretar.
    \end{alertblock}
    \begin{block}{Puntos a considerar}
        \footnotesize
        \begin{itemize}
            \item Todos los datos vienen de \textbf{una sola fuente} (EEDB/ICAO). El estudio no usa datos de vuelo real (FDR).
            \item La prueba de generalización usa solo un año más (2021) de la misma base.
            \item Los datos de 2019 tienen \textbf{valores atípicos} de presión y temperatura ambiente. El artículo no los corrige.
            \item El estudio no compara con modelos más recientes, como redes profundas o ensambles (\textit{ensembles}).
        \end{itemize}
    \end{block}
    \footnotesize Estos puntos no invalidan el estudio. Son líneas de trabajo futuro.
\end{frame}
