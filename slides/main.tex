\section{Contexto y motivación}
%============================================================

\begin{frame}{¿Por qué importa el desempeño de un motor en el despegue?}
    \begin{columns}[c]
        \column{6.7cm}
        \justifying
        Durante el despegue, los motores de un avión trabajan a su \textbf{velocidad y temperatura máximas} para generar el empuje necesario. Este es el punto de operación más exigente de todo el vuelo.
        \bigskip

        Con el uso y el tiempo, el motor se \textbf{degrada}, es decir, deja de comportarse como cuando era nuevo, debido a:
        \begin{itemize}
            \item \textit{Fouling} (ensuciamiento de álabes),
            \item Erosión y abrasión,
            \item Corrosión y oxidación en caliente,
            \item Degradación mecánica general.
        \end{itemize}
        \column{5cm}
        \begin{block}{Consecuencia clave}
            Un motor degradado \textbf{consume más combustible} y \textbf{genera más emisiones contaminantes} (CO$_2$, NO$_x$, hollín, etc.) para producir el mismo empuje.
        \end{block}
    \end{columns}
\end{frame}

\begin{frame}{Objetivo del artículo}
    \justifying
    \Large
    El artículo construye un modelo de \textit{Machine Learning} que predice el \textbf{\textit{fuel flow T/O}} (flujo de combustible en despegue, en kg/s) a partir de variables fáciles de medir. El modelo sirve como indicador indirecto del estado del motor.
\end{frame}

\begin{frame}{¿Por qué el \textit{fuel flow T/O} como indicador de desempeño?}
    \justifying
    \large
    \begin{itemize}
        \item Se usa directamente para calcular las \textbf{emisiones} del motor (junto con los índices de emisión).
        \item \textbf{Cambia} cuando hay fallas o degradación: separación del motor, apagado en vuelo, daño severo, problemas de control de combustible, fugas, etc.
        \item Está \textbf{disponible} para todos los motores certificados en el EEDB.
    \end{itemize}
\end{frame}

\section{Fundamentos de Machine Learning}
%============================================================

\begin{frame}{¿Qué es un problema de regresión supervisada?}
    \justifying
    Antes de presentar los tres métodos del artículo, conviene aclarar la tarea que todos resuelven: es un problema de \textbf{regresión supervisada}.

    \begin{itemize}
        \item Existen ejemplos pasados donde \textbf{se conocen} tanto las variables de entrada (edad del diseño del motor, condiciones ambientales, etc.) como el valor real de la variable a predecir (\textit{fuel flow T/O}).
        \item El algoritmo \textbf{aprende} una función $f$ tal que $\hat{y} = f(x_1, x_2, \dots, x_n)$ se parezca lo más posible al valor real $y$.
        \item Como la salida es un \textbf{número continuo} (kg/s) y no una categoría, es \textbf{regresión} y no clasificación.
    \end{itemize}
    \bigskip

    Los tres modelos del artículo —\textbf{GPR}, \textbf{SVM} y \textbf{MLP}— son tres formas distintas de construir esa función $f$ a partir de los datos.
\end{frame}

\subsection{Regresión de Procesos Gaussianos (GPR)}

\begin{frame}{GPR: la idea intuitiva}
    \justifying
    La \textbf{Regresión de Procesos Gaussianos} (\textit{Gaussian Process Regression}, GPR) no busca ``una sola curva óptima'', sino que trabaja con una \textbf{distribución de probabilidad sobre funciones posibles}.

    \begin{itemize}
        \item Antes de ver datos, el modelo considera \textbf{muchas} funciones plausibles.
        \item Cada punto de dato observado \textbf{descarta} las funciones que no pasan cerca de él.
        \item El resultado es una función media (la predicción) \textbf{más una banda de incertidumbre}: el modelo también indica \textit{qué tan seguro está} de su predicción en cada zona.
    \end{itemize}
    \bigskip
    \footnotesize El artículo usa esa banda para construir los \textbf{intervalos de confianza} que detectan degradación del motor (Sección de resultados).
\end{frame}

\begin{frame}{GPR: función media, incertidumbre y kernel}
    \begin{columns}[c]
        \column{5.6cm}
        \small
        Formalmente (Ec.~3 del artículo):
        $$ f(x) \sim \text{GPR}\big(m(x), k(x,x')\big) $$
        \begin{itemize}
            \item $m(x)$: función \textbf{media}, la mejor predicción puntual.
            \item $k(x,x')$: función de \textbf{covarianza} (\textit{kernel}): mide cuán ``parecidos'' son dos puntos $x$ y $x'$; define qué tan suave es la incertidumbre.
        \end{itemize}
        \bigskip
        El artículo probó 3 \textit{kernels}: \textbf{exponencial}, \textbf{exponencial cuadrado} y \textbf{racional cuadrático}.
        \column{5.6cm}
        \begin{tikzpicture}[scale=0.78]

\end{frame}

\subsection{Máquinas de Vectores de Soporte (SVM)}

\begin{frame}{SVM: la idea intuitiva}
    \justifying\small
    Las \textbf{Máquinas de Vectores de Soporte} (\textit{Support Vector Machines}, SVM) nacieron para \textbf{clasificar}: separar dos grupos con la mejor frontera posible. También sirven para \textbf{regresión}: en vez de separar clases, buscan la función que mejor se ajusta a los datos.
    \medskip

    \begin{itemize}
        \item Se define un \textbf{tubo} de ancho $\varepsilon$ alrededor de la función ajustada.
        \item Los puntos que caen \textbf{dentro} del tubo no penalizan al modelo (se toleran como ruido normal).
        \item Los puntos que quedan \textbf{fuera} del tubo son los que realmente definen la forma de la función: se llaman \textbf{vectores de soporte}.
    \end{itemize}
    \medskip
    \begin{alertblock}{Nota sobre el nombre}
        En tareas de regresión, esta técnica a veces se llama \textit{Support Vector Regression} (SVR). El artículo de Kurt la llama \textbf{SVM}, y esta presentación mantiene ese nombre por consistencia con la fuente.
    \end{alertblock}
\end{frame}

\begin{frame}{SVM: el tubo $\varepsilon$-insensible y los vectores de soporte}
    \begin{columns}[c]
        \column{5.6cm}
        \begin{tikzpicture}[scale=0.78]
        \begin{axis}[
            width=6.6cm, height=5.3cm,
            xlabel={$x$}, ylabel={$y$},
            xtick=\empty, ytick=\empty,
            axis lines=left,
            ]
            \addplot[name path=up, draw=none, domain=0:6] {1.0+0.28*x+0.45};
            \addplot[name path=lo, draw=none, domain=0:6] {1.0+0.28*x-0.45};
            \addplot[suda!18] fill between[of=up and lo];
            \addplot[suda, thick, domain=0:6] {1.0+0.28*x};
            \addplot[only marks, mark=*, black, mark size=1.6pt] coordinates {(0.5,1.3)(1.6,1.55)(2.5,1.75)(3.6,2.0)(4.7,2.15)};
            \addplot[only marks, mark=o, draw=suda, thick, mark size=2.4pt] coordinates {(1.0,2.35)(3.0,1.15)(5.2,2.9)};
        \end{axis}
        \end{tikzpicture}
        \column{5.6cm}
        \footnotesize
        \begin{itemize}
            \item \textbf{Puntos negros}: dentro del tubo $\pm\varepsilon$, no penalizan.
            \item \textbf{Círculos rojos}: fuera del tubo, son los \textbf{vectores de soporte} que ``sostienen'' la forma de la función.
            \item Si los datos no son lineales, se usa una \textbf{función kernel} para proyectarlos a un espacio donde sí lo sean (\textit{kernel trick}).
        \end{itemize}
        \footnotesize\textit{Esquema ilustrativo.} El artículo probó \textit{kernels} \textbf{lineal}, \textbf{cuadrático} y \textbf{cúbico}.
    \end{columns}
\end{frame}

\subsection{Perceptrón Multicapa (MLP)}

\begin{frame}{MLP: la neurona artificial y la red}
    \justifying
    El \textbf{Perceptrón Multicapa} (\textit{Multilayer Perceptron}, MLP) es un tipo de \textbf{red neuronal artificial}, inspirada (de forma muy simplificada) en cómo se conectan las neuronas biológicas.

    \begin{itemize}
        \item Cada \textbf{neurona artificial} recibe varias entradas, las multiplica por \textbf{pesos}, las suma, y aplica una \textbf{función de activación} no lineal (p. ej. \textit{tansig}, \textit{logsig}, \textit{purelin}).
        \item Las neuronas se organizan en \textbf{capas}: capa de entrada, una o más capas \textbf{ocultas}, y capa de salida.
        \item El modelo \textbf{aprende} ajustando los pesos mediante \textbf{retropropagación} (\textit{backpropagation}): compara la predicción con el valor real y corrige los pesos hacia atrás, capa por capa.
    \end{itemize}
\end{frame}

\begin{frame}{Selección de variables de entrada}
    \justifying\small
    Antes de entrenar los modelos de ML, el autor hizo un \textbf{análisis de regresión múltiple} para confirmar cuáles variables realmente influyen en el \textit{fuel flow T/O}.
    \bigskip

    \begin{columns}[c]
    \column{5.6cm}
    \begin{block}{Resultado del análisis (Tablas 7 y 8)}
        \begin{itemize}
            \item $R = 0.990$, $R^2 = 0.979$ (el modelo explica el 97.9\% de la variación),
            \item Prueba ANOVA: Sig. $= 0.000 < 0.001$ $\Rightarrow$ el modelo es \textbf{altamente significativo}.
        \end{itemize}
    \end{block}
    \column{5.6cm}
    \begin{alertblock}{7 variables seleccionadas}
        Tipo de motor, \textit{bypass ratio}, \textit{press ratio}, \textit{rated output}, presión ambiente, temperatura ambiente y humedad ambiente.
    \end{alertblock}
    \end{columns}
    \bigskip
    Estas 7 variables son las que después alimentan a GPR, SVM y MLP como entradas (ver diagramas de las Figuras~4 y~5 del artículo).
\end{frame}

\begin{frame}{La ecuación de regresión múltiple resultante}
    \small
    $$ y = \beta_0 + \beta_1 p_1 + \beta_2 p_2 + \cdots + \beta_n p_n + \epsilon \quad \text{(Ec.~1)} $$
    Con los coeficientes estimados (Tabla 9), la ecuación queda (Ec.~2):
    \begin{align*}
        \text{Fuel Flow } T/O = \ & 1.302 - 0.49 \times \text{tipo de motor} - 0.1 \times \text{bypass ratio} \\
        & - 7.308\times10^{-5} \times \text{press ratio} + 0.009 \times \text{rated output} \\
        & + 2.242\times10^{-7} \times \text{ambient baro} - 0.002 \times \text{ambient temp} \\
        & - 4.612 \times \text{ambient humidity}
    \end{align*}
    \bigskip
    \footnotesize\textbf{Lectura rápida de signos:} a mayor \textit{rated output} (motor más potente), mayor flujo de combustible ($\beta>0$); a mayor \textit{bypass ratio} (motores más eficientes/modernos), menor flujo de combustible ($\beta<0$). Esta regresión \textbf{no es} uno de los tres modelos de ML: solo sirvió para \textbf{validar} qué variables usar como entrada.
\end{frame}

\begin{frame}{Métricas para comparar los modelos}
    \justifying\small
    Los tres modelos se comparan con las mismas tres métricas de error (más bajo = mejor), calculadas sobre $n$ observaciones, donde $FF_{T/O}$ es el valor real y $ff_{T/O}$ el valor predicho:
    \begin{align*}
        \text{MSE} &= \frac{1}{n}\sum_{i=1}^n (FF_{T/O} - ff_{T/O})^2 & \text{(Ec.~5, error cuadrático medio)}\\
        \text{MAE} &= \frac{1}{n}\sum_{i=1}^n |FF_{T/O} - ff_{T/O}| & \text{(Ec.~6, error absoluto medio)}\\
        \text{MAPE} &= \frac{100}{n}\sum_{i=1}^n \left| \frac{FF_{T/O} - ff_{T/O}}{FF_{T/O}} \right| & \text{(Ec.~7, error porcentual absoluto medio)}
    \end{align*}
    \begin{alertblock}{Criterio de calidad (Lewis, 1982)}
        Un modelo con \textbf{MAPE $< 10\%$} se considera de \textbf{alta precisión}; los autores citan que valores por debajo de ese umbral se consideran un modelo ``muy bueno''.
    \end{alertblock}
\end{frame}

\section{Conclusiones y cierre}
%============================================================

\begin{frame}{Hallazgos principales}
    \justifying
    \begin{itemize}
        \item El \textbf{MLP} (modelo 1, entrenado con \textit{trainlm}) fue el mejor modelo global: mejor desempeño en validación y prueba, con MAPE $<5\%$ en todos los conjuntos.
        \item El \textbf{GPR exponencial} logró el mejor ajuste en entrenamiento, pero con más error en validación/prueba que el MLP.
        \item El \textbf{SVM cuadrático} fue el mejor de los tres \textit{kernels} de SVM, pero tuvo los errores más altos de los tres modelos finalistas.
        \item El MLP \textbf{generalizó bien} a datos de un año distinto (2021), con MAPE de solo 2.90\%.
        \item Se construyó un sistema de \textbf{intervalos de confianza (99\%)} que permite señalar posibles degradaciones sin desarmar el motor.
    \end{itemize}
\end{frame}

\begin{frame}{Aporte práctico del estudio}
    \justifying
    \begin{block}{¿Qué permite este modelo que antes no era posible?}
        Evaluar el desempeño de un motor en el despegue \textbf{sin desmontarlo} y \textbf{sin requerir un experto}, usando solo datos ya disponibles en el EEDB.
    \end{block}
    \bigskip
    Esto puede ayudar a reducir:
    \begin{itemize}
        \item El consumo de combustible innecesario,
        \item Los costos de mantenimiento (al detectar degradación temprano),
        \item El costo total del vuelo,
        \item El riesgo de accidentes asociado a fallas no detectadas,
        \item Las emisiones contaminantes de la aviación.
    \end{itemize}
\end{frame}

\begin{frame}{Discusión crítica (reflexión propia)}
    \justifying\small
    \begin{alertblock}{Fortalezas}
        Comparación rigurosa y transparente de 3 familias de modelos con la misma partición de datos; validación real con un segundo dataset (2021); uso de intervalos de confianza con una interpretación práctica clara.
    \end{alertblock}
    \begin{block}{Puntos a considerar}
        \begin{itemize}
            \item Los datos provienen de una \textbf{única fuente institucional} (EEDB/ICAO); no se prueba con datos de otras bases o de vuelo real (FDR).
            \item La validación de generalización usa un solo año adicional (2021), de la misma base de datos.
            \item No se comparan con modelos de \textit{deep learning} más recientes (p. ej. redes profundas, \textit{ensembles}), que otros trabajos de la Tabla 3 sí exploran.
        \end{itemize}
    \end{block}
    \footnotesize Estos puntos no invalidan el estudio; son posibles líneas de trabajo futuro.
\end{frame}
